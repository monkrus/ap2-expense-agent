QUICK START COMMANDS
================================================================================

1. Generate JWT Secret
   python -c "import secrets; print(secrets.token_urlsafe(64))"

2. Create Cloud SQL Instance
   gcloud sql instances create ap2-expenses \
     --database-version=POSTGRES_11 --region=us-central1 \
     --tier=db-f1-micro

3. Create Database
   gcloud sql databases create expenses --instance=ap2-expenses

4. Set Environment Variables
   export ENVIRONMENT=production
   export DEBUG=False
   export JWT_SECRET=<generated_secret>
   export DATABASE_URL=postgresql://user:pass@...
   export CORS_ORIGINS=https://app.yourdomain.com

5. Run Startup Validation
   cd backend && python -c "from src.startup_checks import validate_settings; validate_settings()"

6. Run Tests
   cd backend && pytest tests/test_auth.py -v

7. Deploy to Cloud Run
   gcloud run deploy ap2-backend \
     --image=gcr.io/PROJECT_ID/ap2-backend:latest \
     --platform=managed --region=us-central1 \
     --memory=2Gi --cpu=2 \
     --set-env-vars ENVIRONMENT=production,DEBUG=False
